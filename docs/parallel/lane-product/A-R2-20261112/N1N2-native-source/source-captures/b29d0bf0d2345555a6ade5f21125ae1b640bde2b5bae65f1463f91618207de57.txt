 /** Register the Host's own workspace ref for an issued view.
  *
  * The ref is a name the Host generated; this service is what gives it meaning, and only
  * for the exact view it issued. The complete binding is snapshotted here, privately:
  * the verified owner, the project, the committed transaction, the snapshot, the
  * revision, the source digest, the native project id and the generation of this copy.
  *
  * The generation is the *service's* fact: it comes from the exact issued view, and a
  * caller cannot state it, guess it or prime it. A ref is issued once and is never
  * rebound. */
 registerWorkspaceRef(view:PreparedNativeWorkspace,ref:string,registration:Omit<NativeWorkspaceBinding,'generation'>):NativeWorkspaceBinding{
  if(typeof ref!=='string'||!/^[A-Za-z0-9][A-Za-z0-9_-]{0,127}$/.test(ref))throw conflict();
  if(this.#closing)throw unavailable();
  const record=this.#prepared.get(view.project_id);
  if(!record||record.view!==view)throw conflict();
  if(this.#refs.has(ref))throw conflict();
  const binding=record.binding;
  if(registration.project_id!==binding.project_id||registration.principal_id!==binding.principal_id
   ||registration.transaction_id!==binding.transaction_id||registration.snapshot_id!==binding.snapshot_id
   ||registration.source_digest!==binding.source_digest||registration.revision!==binding.revision)throw conflict();
  if(registration.native_project_id!==null&&(typeof registration.native_project_id!=='string'||registration.native_project_id.length===0))throw conflict();
  const snapshot=Object.freeze({project_id:registration.project_id,principal_id:registration.principal_id,
   transaction_id:registration.transaction_id,snapshot_id:registration.snapshot_id,source_digest:registration.source_digest,
   revision:registration.revision,native_project_id:registration.native_project_id,generation:record.generation});
  this.#refs.set(ref,{record,binding:snapshot,invalidated:false});
  return snapshot;
 }
 /** Record a native cache witness for an issued view.
  *
  * The brand alone is NOT authority: a hand-branded witness, a witness of another
  * service and a witness of another instance are all refused here. What makes a
  * witness real is the private binding of this service: the witness must be one
  * THIS service produced under a production lease THIS service issued, and that
  * lease must still be current.
  *
  * A recorded witness is STAGED, not available: it is not visible to a lookup or
  * to the reader until every resource of the production that made it has settled.
  * The witness is lost with this instance - it is never a persistent authority. */
 record(witness:NativeCacheWitness):boolean{
  if(!isIssuedNativeCacheWitness(witness))return false;
  // The private association: the witness and the production lease of this service.
  const lease=witnessLeases.get(witness);
  if(lease===undefined)return false;
  const facts=productionLeases.get(lease);
  if(facts===undefined||facts.service!==this)return false;
  // The lease must still be current: the owner, the base, the control and the
  // generation of the view it was issued against.
  if(facts.lease.assertCurrent()!==true)return false;
  const registration=this.#refs.get(witness.workspace_ref);
  if(registration===undefined||registration.invalidated)return false;
  if(registration.binding.generation!==witness.generation)return false;
  // The witness must speak about the ISSUED root of that very view: a witness about
  // another tree is refused, however it was made.
  if(witness.root!==registration.record.view.source_root.value)return false;
  try{this.#check(registration.record.binding);}catch{return false;}
  return this.#witnesses.stage(witness);
 }
 /** The witness of this view's key, or null when there is none.
  *
  * A ref that was never issued, an invalidated ref and a generation that is not the
  * issued one have no witness at all. A staged witness - one whose resources have
  * not settled - is not a witness yet, and a revoked one never becomes one. */
 lookup(ref:string,generation:number,key:string):NativeCacheWitness|null{
  const registration=this.#refs.get(ref);
  if(registration===undefined||registration.invalidated)return null;
  if(registration.binding.generation!==generation)return null;
  const witness=this.#witnesses.lookup(ref,generation,key);
  if(witness===null)return null;
  // Only a witness THIS service produced under its own production lease is
  // visible here: the brand alone decides nothing.
  const lease=witnessLeases.get(witness);
  if(lease===undefined)return null;
  const facts=productionLeases.get(lease);
  if(facts===undefined||facts.service!==this)return null;
  return witness;
 }
 /** The witness of this view's key, or null when there is none. */
 cacheWitness(ref:string,generation:number,key:string):NativeCacheWitness|null{
  return this.lookup(ref,generation,key);
 }
 /** The witness that authorised ONE output of this view, or null when there is none.
  *
  * It is the CURRENT authorisation of that output: the draw that really produced it
  * last. An output whose current authorisation was revoked or invalidated has none
  * at all - the older authorisation of the same output does not come back. */
 cacheWitnessByOutput(ref:string,generation:number,output_path:string):NativeCacheWitness|null{
  const registration=this.#refs.get(ref);
  if(registration===undefined||registration.invalidated)return null;
  if(registration.binding.generation!==generation)return null;
  const witness=this.#witnesses.lookupByOutput(ref,generation,output_path);
  if(witness===null)return null;
  const lease=witnessLeases.get(witness);
  if(lease===undefined)return null;
  const facts=productionLeases.get(lease);
  if(facts===undefined||facts.service!==this)return null;
  return witness;
 }
 /** Acquire the production lease of an issued view.
  *
  * It is the authority to produce inside that view, and it is issued HERE and
  * nowhere else: the ref must be registered and still current, the expectation
  * must name the very binding the ref was registered with, the owner, the base and
  * the control must still be the ones of the committed source, and the view must
  * still be the issued one.
  *
  * The lease is registered in this module's private table: a copy of it, a JSON
  * object and a lease of another service are not in that table at all, and they
  * are refused before any producer runs. */
 acquireCacheProductionLease(ref:string,expectation:NativeWorkspaceExpectation):NativeCacheProductionLease{
  const registration=this.#refs.get(ref);
  if(!registration||registration.invalidated)throw unavailable();
  const binding=registration.binding;
  if(expectation.project_id!==binding.project_id||expectation.snapshot_id!==binding.snapshot_id
   ||expectation.source_digest!==binding.source_digest||expectation.revision!==binding.revision
   ||expectation.native_project_id!==binding.native_project_id)throw conflict();
  // The owner, the base and the control of the committed source.
  this.#check(registration.record.binding);
  // The view must still be the issued one.
  if(this.#prepared.get(binding.project_id)!==registration.record)throw conflict();
  if(!this.isPrepared(registration.record.view))throw conflict();
  const lease=this.acquireExpectedLease(ref,expectation);
  const production=Object.freeze({workspace_ref:ref,generation:binding.generation,root:registration.record.view.source_root.value});
  productionLeases.set(production,{workspace_ref:ref,generation:binding.generation,root:registration.record.view.source_root.value,
   lease,pool:this.#descriptorPool,service:this});
  return production;
 }
 /** The serving owner THIS service registered for the issued view of a production.
  *
  * The service creates it and keeps it: the owner holds its own listener, its own
  * handler and the identity of the ONE root it serves, and it is registered against
  * the exact issued view - not against a number that happens to be equal somewhere
  * else, and not against a string a caller hands over.
  *
  * The origin of the draw is taken from the owner's own private facts, so the page
  * the producer draws is served from the very working copy the Host issued. */
 async servingFor(production:NativeCacheProductionLease,producer:NativeCacheProducer,
  options?:{headers?:Record<string,string>}):Promise<OpenFilmServingOwner>{
  if(this.#closing)throw unavailable();
  const facts=productionLeases.get(production as unknown as object);
  if(facts===undefined||facts.service!==this)refuseCache('NATIVE_CACHE_PRODUCTION_FOREIGN','The production lease is not one this service issued');
  if(facts.lease.assertCurrent()!==true)refuseCache('NATIVE_CACHE_PRODUCTION_STALE','The issued view is no longer current');
  if(!isIssuedOpenFilmCacheProducer(producer))refuseCache('NATIVE_CACHE_PRODUCER_FOREIGN','The cache producer is not the fixed one this Host issued');
  const record=this.#prepared.get(facts.lease.project_id);
  if(record===undefined||record.view.source_root.value!==facts.root)refuseCache('NATIVE_CACHE_PRODUCTION_STALE','The issued view is no longer current');
  const existing=this.#servings.get(record);
  if(existing!==undefined)return existing.owner;
  const owner=await createPinnedOpenFilmServingOwner(producer,facts.root,options);
  const binding=openFilmServingBinding(owner);
  if(binding===null)refuseCache('NATIVE_CACHE_ORIGIN_FOREIGN','The serving owner is not one this Host issued');
  // The serving must serve the EXACT working copy this service still holds: the
  // same canonical path AND the same directory - device and inode - that the held
  // descriptor of the issued view names.
  const held=fstatSync(record.directory.fd,{bigint:true});
  if(binding.canonical_root!==realpathSync(facts.root)
   ||binding.dev!==held.dev.toString()||binding.ino!==held.ino.toString())
   refuseCache('NATIVE_CACHE_ORIGIN_FOREIGN','The serving owner does not serve the issued root');
  this.#servings.set(record,{owner,generation:binding.generation,canonical_root:binding.canonical_root,dev:binding.dev,ino:binding.ino});
  this.#servingOwners.add(owner);
  return owner;
 }
 /** Produce ONE native cache entry with the FIXED producer, and witness it.
  *
  * The order is fixed and it is the whole point of the witness:
  *  1. the production lease must be THIS service's, and it must be real: an
  *     unknown object, a copy and a lease of another service are refused BEFORE
  *     the producer is called, so a capability that is not ours never has a side
  *     effect;
  *  2. the lease must still be current: the owner, the base, the control and the
  *     generation of the view it was issued against;
  *  3. the producer must be the one the Host's own factory issued - a lookalike
  *     that merely has the right shape is refused;
  *  4. the serving must be the one THIS service registered for THIS issued view:
  *     the same object, still alive, still serving the very directory the service
  *     holds - so the page the producer draws comes from the issued root and from
  *     nowhere else;
  *  5. the request must be the exact request of its family: the film poster is
  *     the paired output of the film page at -1 and 480, and a page frame that
  *     would write that poster is the poster request, not a page frame;
  *  6. the key of the request is computed with the upstream's own formula and the
  *     upstream's own overrides predicate;
  *  7. a pre-existing, unwitnessed member of that key - and the poster this call
  *     would overwrite - are REFUSED: a cache hit is not proof that the producer
  *     drew the picture, and the old bytes stay exactly where they are;
  *  8. the tree is RETAINED before the producer runs, so the state of the page it
  *     draws from is known before the draw;
  *  9. the fixed producer is invoked, once, awaited;
  *  10. the tree is retained again afterwards, and every file the producer really
  *     read is read back through a retained descriptor and compared with the very
  *     identity it had before the draw;
  *  11. the final synchronous fence: the held dependencies, the written files and
  *     the chain above the root are compared by their EXACT identity;
  *  12. the witness is STAGED - it is not visible to anybody yet;
  *  13. every resource of this call is settled, and a failure of that settlement
  *      revokes the witness and latches this service;
  *  14. after the settlement the real path identities and the chain are re-checked
  *      against the identity the closure held, the lease is re-checked against the
  *      real authority, the serving is re-checked against the owner it was
  *      registered with, the LAST identity observation follows that check, and only
  *      then - with no await and no external callback - the witness is activated.
  *
  * The witness is a fact about THIS call. It is never an author source, never an
  * independent expected fact, and it is lost with the workspace instance that holds
  * it. */
 async produceNativeCacheEntry(production:NativeCacheProductionLease,request:{
  readonly producer:NativeCacheProducer;
  readonly serving:OpenFilmServingOwner;readonly rel:string;readonly ms:number;readonly width:number;
  readonly overrides:unknown[]|null;readonly kind:'pageframe'|'film-poster';
 }):Promise<NativeCacheWitness>{
  // A closed service admits NO new production. The production that was already
  // accepted keeps its own resources - its listener, its pages and its probes -
  // until it has really settled.
  if(this.#closing)throw unavailable();
  // The whole production is ONE admitted operation of this service. A close of the
  // service waits for it: the producer, the retained trees and their settlement
  // are never cut in half by a shutdown, and the working copy is never disposed
  // while a producer is still drawing inside it.
  const work=this.#produce(production,request);
  this.#operations.add(work);
  void work.then(()=>this.#operations.delete(work),()=>this.#operations.delete(work));
  return work;
 }
 /** Whether this service can still say what it holds. A failed release latches it. */
 get uncertain():boolean{return this.#uncertain;}
 async #produce(production:NativeCacheProductionLease,request:{
  readonly producer:NativeCacheProducer;
  readonly serving:OpenFilmServingOwner;readonly rel:string;readonly ms:number;readonly width:number;
  readonly overrides:unknown[]|null;readonly kind:'pageframe'|'film-poster';
 }):Promise<NativeCacheWitness>{
  // 1) The lease is THIS service's, and it is real.
  const facts=productionLeases.get(production as unknown as object);
  if(facts===undefined)refuseCache('NATIVE_CACHE_PRODUCTION_FOREIGN','The production lease is not one this service issued');
  if(facts.service!==this)refuseCache('NATIVE_CACHE_PRODUCTION_FOREIGN','The production lease belongs to another workspace service');
  // 2) The view is still the issued one.
  if(facts.lease.assertCurrent()!==true)refuseCache('NATIVE_CACHE_PRODUCTION_STALE','The issued view is no longer current');
  // 3) The producer is the FIXED one.
  if(!isIssuedOpenFilmCacheProducer(request.producer))refuseCache('NATIVE_CACHE_PRODUCER_FOREIGN','The cache producer is not the fixed one this Host issued');
  // A service whose own settlement failed can no longer say what it holds.
  if(this.#uncertain)refuseCache('NATIVE_CACHE_PRODUCTION_UNCERTAIN','This workspace service cannot say what it holds');
  const root=facts.root;
  // 4) The serving is the one THIS service registered for THIS issued view: the very
  //    object, still alive, still serving the very directory the service holds. The
  //    origin of the draw is read from that owner's own private facts - never from a
  //    string of the caller's, and never from a number that is merely equal to
  //    something else. A serving of another root, a copy, a lookalike and a closed
  //    owner are all refused HERE, before the producer runs: no page is requested and
  //    no cache is written.
  const record=this.#prepared.get(facts.lease.project_id);
  if(record===undefined||record.view.source_root.value!==root)refuseCache('NATIVE_CACHE_PRODUCTION_STALE','The issued view is no longer current');
  const registered=this.#servings.get(record);
  if(registered===undefined)refuseCache('NATIVE_CACHE_ORIGIN_FOREIGN','This production has no serving of this service');
  if(request.serving!==registered.owner)refuseCache('NATIVE_CACHE_ORIGIN_FOREIGN','The serving is not the one this service registered');
  const binding=openFilmServingBinding(request.serving);
  if(binding===null||binding.closed||binding.generation!==registered.generation)
   refuseCache('NATIVE_CACHE_ORIGIN_FOREIGN','The serving owner is not the one this Host issued');
  const held=fstatSync(record.directory.fd,{bigint:true});
  if(binding.canonical_root!==registered.canonical_root||binding.dev!==held.dev.toString()||binding.ino!==held.ino.toString())
   refuseCache('NATIVE_CACHE_ORIGIN_FOREIGN','The serving owner does not serve the issued root');
  const origin=binding.origin;
  // 5) and 6) The exact request of the family, and the upstream's own key.
  const width=Math.max(32,Math.min(960,Math.round(request.width/2)*2||160));
  const overridesJson=upstreamCacheTweaks(request.overrides);
  const writesPoster=request.rel==='film.html'&&request.ms<0&&width===480;
  if(request.kind==='film-poster'){
   if(request.rel!=='film.html'||request.ms!==-1||width!==480)
    refuseCache('NATIVE_CACHE_REQUEST_FOREIGN','The film poster is the paired output of the film page at -1 and 480');
  }else if(writesPoster){
   refuseCache('NATIVE_CACHE_REQUEST_FOREIGN','That request writes the film poster: it is the poster request, not a page frame');
  }
  const frameKey=nativeCacheKey({rel:request.rel,ms:request.ms,width,overrides_json:overridesJson});
  const key=request.kind==='film-poster'?'film-poster':frameKey;
  const jpegPath=`.film/cache/pageframes/${frameKey}.jpg`;
  const depsPath=`.film/cache/pageframes/${frameKey}.json`;
  const stampPath=`.film/cache/pageframes/${frameKey}.stamp`;
  const posterPath='.film/cache/film-poster.jpg';
  const durationPath='.film/cache/film-duration.json';
  // 7) An unwitnessed member of that key, and the poster this call would
  //    overwrite, are refused - and their bytes stay where they are.
  for(const member of [jpegPath,depsPath,stampPath])
   if(isRegularFileWithin(root,member))refuseCache('NATIVE_CACHE_PREEXISTING','The native cache entry is already there and was not produced by this call');
  if(request.kind==='film-poster'&&isRegularFileWithin(root,posterPath))
   refuseCache('NATIVE_CACHE_PREEXISTING','The film poster is already there and this call would overwrite it');
  let before:RetainedNativeClosure|undefined;
  let after:RetainedNativeClosure|undefined;
  let primary:unknown;
  let witness:NativeCacheWitness|undefined;
  let chain:readonly {readonly path:string;readonly dev:string;readonly ino:string}[]=[];
  try{
   // 8) The tree is retained BEFORE the draw. The phase is the PRODUCTION's own
   //    word - it is wrapped onto the observation here, so a caller of the
   //    production can never say which phase it is in.
   const phaseOps=(phase:'PRE'|'POST'):RetainedNativeDescriptorOps=>
    this.#descriptorOps===undefined?{phase}:{...this.#descriptorOps,phase};
   before=await RetainedNativeClosure.open(root,{},phaseOps('PRE'),facts.pool);
   const beforeFacts=new Map<string,NativeCacheFileIdentity>();
   for(const path of before.fileMembership()){
    const file=await before.retainFile(path);
    beforeFacts.set(path,identityOf(file,path));
   }
   if(!beforeFacts.has(request.rel))refuseCache('NATIVE_CACHE_REQUEST_FOREIGN','The page the request names is not a member of the issued tree');
   // 9) The fixed producer really runs, once, awaited - against the origin of the
   //    serving THIS service registered.
   const produced=await request.producer.pageFrame({root,base:origin,rel:request.rel,ms:request.ms,width,overrides:request.overrides});
   // 10) The tree is retained again AFTER the draw - and it is a tree of its own:
   //     a different phase, a different namespace, its own allocations.
   after=await RetainedNativeClosure.open(root,{},phaseOps('POST'),facts.pool);
   const depsFile=await after.retainFile(depsPath);
   const depsText=await after.readText(depsFile);
   let parsed:unknown;
   try{parsed=JSON.parse(depsText);}catch{refuseCache('NATIVE_CACHE_OUTPUT_MALFORMED','The dependency list of the cache entry is not JSON');}
   const declared=(parsed as {files?:unknown}).files;
   if(!Array.isArray(declared)||declared.length===0||declared.some(file=>typeof file!=='string'))
    refuseCache('NATIVE_CACHE_OUTPUT_MALFORMED','The dependency list of the cache entry names no file');
   const dependencies:NativeCacheFileIdentity[]=[];
   for(const dependency of declared as string[]){
    const file=await after.retainFile(dependency);
    const identity=identityOf(file,dependency);
    const known=beforeFacts.get(dependency);
    if(known===undefined)refuseCache('NATIVE_CACHE_DEPENDENCY_UNRELATED','The producer read a file that was not part of the issued tree');
    // The EXACT identity: the same device, the same inode, the same bytes, the
    // same nanosecond change times and the same number of names. A file that was
    // replaced by an equal copy is refused here.
    if(known.dev!==identity.dev||known.ino!==identity.ino||known.sha256!==identity.sha256
     ||known.bytes!==identity.bytes||known.mtime_ns!==identity.mtime_ns||known.ctime_ns!==identity.ctime_ns
     ||known.nlink!==identity.nlink)
     refuseCache('NATIVE_CACHE_SOURCE_RACED','A file the producer read changed while it was drawing');
    dependencies.push(identity);
   }
   if(!dependencies.some(entry=>entry.path===request.rel))refuseCache('NATIVE_CACHE_DEPENDENCY_UNRELATED','The producer did not read the page the request names');
   const stampFile=await after.retainFile(stampPath);
   const stamp=await after.readText(stampFile);
   if(nativeCacheStamp(dependencies)!==stamp)refuseCache('NATIVE_CACHE_SOURCE_RACED','A file the producer read changed while it was drawing');
   const picture=await after.retainFile(jpegPath);
   const producedDigest=createHash('sha256').update(produced).digest('hex');
   if(producedDigest!==picture.sha256)
    refuseCache('NATIVE_CACHE_OUTPUT_MISMATCH','The picture the producer returned is not the file that was written');
   const written:NativeCacheFileIdentity[]=[identityOf(picture,jpegPath),identityOf(depsFile,depsPath),identityOf(stampFile,stampPath)];
   const retained=new Map<string,RetainedNativeFile>([[jpegPath,picture],[depsPath,depsFile],[stampPath,stampFile]]);
   let posterSha:string|null=null;
   if(request.kind==='film-poster'){
    if(!isRegularFileWithin(root,posterPath))refuseCache('NATIVE_CACHE_OUTPUT_MISSING','The producer did not write the poster it must write');
    const poster=await after.retainFile(posterPath);
    // The poster is the paired output of THIS draw: it is the very picture this
    // call returned, and nothing else.
    if(poster.sha256!==producedDigest)
     refuseCache('NATIVE_CACHE_OUTPUT_MISMATCH','The poster is not the picture of this call');
    posterSha=poster.sha256;
    written.push(identityOf(poster,posterPath));
    retained.set(posterPath,poster);
   }
   // The FILM DURATION is another REAL output of this draw: the upstream writes it
   // while it draws the film's own page. It is captured HERE, from the tree this
   // call really holds after the draw, and it enters the PENDING witness as an
   // output of that draw - so it is proved by the same final identity fence as every
   // other output, and it becomes visible only when everything has settled.
   // A duration that no witnessed draw produced is not a duration at all: the reader
   // refuses it, and the ledger never falls back to an older authorisation for it.
   if(isRegularFileWithin(root,durationPath)){
    const duration=await after.retainFile(durationPath);
    written.push(identityOf(duration,durationPath));
    retained.set(durationPath,duration);
   }
   // 11) The final synchronous fence, with the closures still held: the exact
   //     identity of every file the producer read and wrote, and the chain above
   //     the root. No await and no callback happen inside it.
   for(const [path,file] of retained)if(!after.verifyFileExact(file))refuseCache('NATIVE_CACHE_OUTPUT_MOVED','The cache entry changed while it was produced');
   for(const dependency of dependencies){
    const file=await after.retainFile(dependency.path);
    if(!after.verifyFileExact(file))refuseCache('NATIVE_CACHE_SOURCE_RACED','A file the producer read changed while it was drawing');
   }
   if(!after.verifyRootChain())refuseCache('NATIVE_CACHE_ROOT_MOVED','The issued root moved while the producer was drawing');
   if(!after.verifyDirectoryChain())refuseCache('NATIVE_CACHE_ROOT_MOVED','The parent chain of the cache entry moved while the producer was drawing');
   chain=after.chainIdentities();
   // 12) The witness is STAGED: it is not visible to a lookup or to the reader.
   const candidate:NativeCacheWitness=issueNativeCacheWitness({
    kind:request.kind,workspace_ref:facts.workspace_ref,generation:facts.generation,root,
    rel:request.rel,ms:request.ms,width,overrides_json:overridesJson,key,frame_key:frameKey,
    jpeg_sha256:picture.sha256,deps_sha256:depsFile.sha256,stamp_sha256:stampFile.sha256,poster_sha256:posterSha,
    dependencies,outputs:written,issued_at:new Date().toISOString()});
   witness=candidate;
   witnessLeases.set(candidate,production);
   if(!this.record(candidate))refuseCache('NATIVE_CACHE_WITNESS_REFUSED','The workspace instance refused the witness');
  }catch(error){primary=error;}
  // 13) Every resource of this call is settled, and every failure is kept.
  const cleanupFailures:unknown[]=[];
  if(after)try{await after.release();}catch(error){cleanupFailures.push(error);}
  if(before)try{await before.release();}catch(error){cleanupFailures.push(error);}
  if(primary!==undefined){
   // The primary keeps its place and the cleanup failures are merged into it.
   if(witness!==undefined)this.#witnesses.revoke(witness);
   // A release that did not finish latches this service, WHETHER the producer
   // itself failed or not: the failure of the producer says nothing about
   // whether the resources are gone. The genuine release failures the failure
   // already carries are counted too - they are the failures of the OPEN of the
   // retained tree itself.
   const genuine=[...cleanupFailures,...retainedCleanupFailures(primary)];
   if(genuine.length>0)this.#uncertain=true;
   if(cleanupFailures.length>0)mergeRetainedCleanupCauses(primary,cleanupFailures);
   throw primary;
  }
  if(cleanupFailures.length>0){
   // The production happened, but its resources did not settle: nothing is
   // delivered, the staged witness is revoked, and this service can no longer say
   // what it holds.
   if(witness!==undefined)this.#witnesses.revoke(witness);
   this.#uncertain=true;
   const failure=mergeRetainedCleanupCauses(new NativeCacheWitnessError('NATIVE_CACHE_CLEANUP_FAILED','The native cache production could not release everything'),cleanupFailures);
   throw failure;
  }
  if(witness===undefined)refuseCache('NATIVE_CACHE_WITNESS_REFUSED','The production produced no witness');
  // 14) The settlement is done and the witness is still PENDING. The real path
  //     identities of the outputs AND of the dependencies, and the whole parent
  //     chain, are re-checked against the identity the closure held; the lease is
  //     re-checked against the real authority; the LAST identity observation
  //     follows that check; and only then - with no await and no external
  //     callback - the witness is activated.
  const checkFiles=(identities:readonly NativeCacheFileIdentity[],code:string,message:string):void=>{
   for(const identity of identities){
    const current=pathIdentity(join(root,identity.path));
    if(current===null)refuseCache(code,message);
    if(!sameIdentity(current,{dev:identity.dev,ino:identity.ino,size:String(identity.bytes),
     mtime_ns:identity.mtime_ns,ctime_ns:identity.ctime_ns,nlink:identity.nlink}))
     refuseCache(code,message);
   }
  };
  const checkOutputs=():void=>checkFiles(witness!.outputs,'NATIVE_CACHE_OUTPUT_MOVED','The cache entry changed while it was produced');
  const checkDependencies=():void=>checkFiles(witness!.dependencies,'NATIVE_CACHE_SOURCE_RACED','A file the producer read changed while it was produced');
  const checkChain=():void=>{
   for(const entry of chain){
    let info;
    try{info=lstatSync(entry.path,{bigint:true});}catch{refuseCache('NATIVE_CACHE_ROOT_MOVED','The chain above the issued root moved');}
    if(!info.isDirectory()||info.isSymbolicLink()||info.dev.toString()!==entry.dev||info.ino.toString()!==entry.ino)
     refuseCache('NATIVE_CACHE_ROOT_MOVED','The chain above the issued root moved');
   }
  };
  try{
   checkChain();
   checkOutputs();
   checkDependencies();
   // The lease is re-checked against the REAL authority: the owner, the base, the
   // control and the generation of the view.
   if(facts.lease.assertCurrent()!==true)refuseCache('NATIVE_CACHE_PRODUCTION_STALE','The issued view is no longer current');
   // The serving is re-checked against the owner THIS service registered: the same
   // object, still alive, still serving the very directory the service holds. A
   // serving that was closed, replaced or rebound while the producer was drawing
   // never activates a witness.
   const live=openFilmServingBinding(request.serving);
   if(live===null||live.closed||live.generation!==registered.generation
    ||live.canonical_root!==registered.canonical_root||live.origin!==registered.owner.origin)
    refuseCache('NATIVE_CACHE_ORIGIN_FOREIGN','The serving owner changed while the producer was drawing');
   const liveHeld=fstatSync(record.directory.fd,{bigint:true});
   if(live.dev!==liveHeld.dev.toString()||live.ino!==liveHeld.ino.toString())
    refuseCache('NATIVE_CACHE_ORIGIN_FOREIGN','The serving owner does not serve the issued root');
   // The LAST synchronous identity observation, after the last possible callback.
   checkChain();
   checkOutputs();
   checkDependencies();
   if(!this.#witnesses.activate(witness))refuseCache('NATIVE_CACHE_WITNESS_REFUSED','The workspace instance refused the witness');
  }catch(error){
   // The witness never became available: it is revoked, and a revoked witness is
   // never revived.
   this.#witnesses.revoke(witness);
   throw error;
  }
  return witness;
 }

 /** The root of an issued view, or null when there is none. */
 #issuedRootOf(ref:string,generation:number):string|null{
  const registration=this.#refs.get(ref);
  if(registration===undefined||registration.invalidated)return null;
  if(registration.binding.generation!==generation)return null;

 }
 /** Acquire the lease of an issued workspace ref.
  *
  * The caller states the binding it expects; every field it states must be the one the
  * ref was registered with. A ref of another instance, an unknown ref, an invalidated
  * ref and a ref whose view was replaced are all refusals.
  *
  * Every acquisition is its own lease. Releasing one lease releases *that* lease and
  * nothing else: the ref and the other leases of it stay usable, so a proof can be
  * released and a later stable round can prove again on the same view. */
 acquireExpectedLease(ref:string,expectation:NativeWorkspaceExpectation):NativeWorkspaceLease{
  const registration=this.#refs.get(ref);
  if(!registration||registration.invalidated)throw unavailable();
  const binding=registration.binding;
  if(expectation.project_id!==binding.project_id||expectation.snapshot_id!==binding.snapshot_id
   ||expectation.source_digest!==binding.source_digest||expectation.revision!==binding.revision
   ||expectation.native_project_id!==binding.native_project_id)throw conflict();
  const facts:LeaseFacts={registration,released:false};
  const lease=Object.freeze({workspace_ref:ref,project_id:binding.project_id,principal_id:binding.principal_id,
   transaction_id:binding.transaction_id,snapshot_id:binding.snapshot_id,source_digest:binding.source_digest,
   revision:binding.revision,native_project_id:binding.native_project_id,generation:binding.generation,
   root:registration.record.view.source_root.value,
   assertCurrent:()=>this.#assertLeaseCurrent(facts),
   release:()=>{facts.released=true;}});
  this.#leases.set(lease,facts);
  return lease;
 }
 /** Whether this instance issued the lease. A copy of a lease and a lease of another
  *  service are both unrecognised here: the brand is not global. */
 isLease(value:unknown):value is NativeWorkspaceLease{
  return !!value&&typeof value==='object'&&this.#leases.has(value);
 }
 #assertLeaseCurrent(facts:LeaseFacts):boolean{
  try{
   if(facts.released)return false;
   const registration=facts.registration;
   if(registration.invalidated)return false;
   const record=registration.record;
   if(this.#prepared.get(record.binding.project_id)!==record)return false;
   this.#check(record.binding);
   return this.isPrepared(record.view);
  }catch{return false;}
 }

  const parts=assertSafeRelativePath(path).split('/'),parents:FileHandle[]=[];let parent=root;
  try{
   for(const name of parts.slice(0,-1)){

   }

   const file=await open(fdPath(parent)+'/'+parts[parts.length-1],flags,0o600);return {file,parents};
  }catch(error){await Promise.all(parents.map(handle=>handle.close()));throw error;}
 }
 async #read(entry:SnapshotEntry,root:FileHandle,write?:FileHandle){
  const {file,parents}=await this.#leaf(root,entry.relative_path,false);
  try{
   const before=await file.stat({bigint:true});if(!before.isFile()||before.size!==BigInt(entry.bytes))throw conflict();
   const hash=createHash('sha256'),buffer=Buffer.allocUnsafe(65536);let offset=0;
   while(offset<entry.bytes){
    if(this.#abort.signal.aborted)throw unavailable();
    const {bytesRead}=await file.read(buffer,0,Math.min(buffer.length,entry.bytes-offset),offset);if(!bytesRead)throw conflict();
    hash.update(buffer.subarray(0,bytesRead));
    if(write){let done=0;while(done<bytesRead){const result=await write.write(buffer,done,bytesRead-done,offset+done);if(!result.bytesWritten)throw unavailable();done+=result.bytesWritten;}}
    offset+=bytesRead;
   }
   const after=await file.stat({bigint:true});
   if(before.dev!==after.dev||before.ino!==after.ino||before.size!==after.size||before.mtimeNs!==after.mtimeNs||before.ctimeNs!==after.ctimeNs||hash.digest('hex')!==entry.sha256)throw conflict();
  }finally{await file.close();await Promise.all(parents.map(handle=>handle.close()));}
 }
 async #copy(entry:SnapshotEntry,source:FileHandle,destination:FileHandle){
  const {file,parents}=await this.#leaf(destination,entry.relative_path,true);
  try{await this.#read(entry,source,file);await file.sync();}
  finally{await file.close();await Promise.all(parents.map(handle=>handle.close()));}
 }
 async #verify(record:HeldWorkspace){
  const root=await this.#assertRoot(),path=fdPath(root)+'/'+record.name,stats=await lstat(path),held=await record.directory.stat();
  if(!stats.isDirectory()||stats.isSymbolicLink()||held.dev!==stats.dev||held.ino!==stats.ino||await realpath(path)!==this.#path+'/'+record.name)throw conflict();
  for(const entry of record.files)await this.#read(entry,record.directory);
 }
 /** Phase one stops cache I/O; native may still be using a previous view until shutdown.
  *  Every ref is invalidated with it: no proof of this service survives its close. */
 close(){this.#closing=true;this.#abort.abort();for(const registration of this.#refs.values())registration.invalidated=true;return this.#stop??=Promise.allSettled([...this.#operations]).then(()=>undefined);}
 /** Called only AFTER native Studio/Gateway have stopped. Never delete a live native root. */
 disposeAfterNative(){return this.#dispose??=(async()=>{await this.close();
  // The serving owners are closed only NOW. An accepted draw may still be asking
  // its pages and its assets for answers, and the listener has to stay up until
  // they have settled - it is never cut in half by a close, and an aborted request
  // is never called a settled one. The roots are released after that.
  const failures:unknown[]=[];
  for(const owner of this.#servingOwners){try{await owner.close();}catch(error){failures.push(error);}}
  this.#servingOwners.clear();this.#servings.clear();
  const root=this.#root;if(!root)return;
  for(const record of this.#held){await record.directory.close();await rm(fdPath(root)+'/'+record.name,{recursive:true,force:true});}
  this.#held.clear();this.#prepared.clear();this.#refs.clear();this.#generations.clear();await root.close();this.#root=undefined;
  if(failures.length>0)throw failures[0];
 })();}
}
