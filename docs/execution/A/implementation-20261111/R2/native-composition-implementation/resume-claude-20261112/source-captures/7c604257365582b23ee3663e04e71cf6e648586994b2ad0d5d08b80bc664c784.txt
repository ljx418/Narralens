/**
 * The N1N2 component cases.
 *
 * The real chain end to end: the real HTTP bootstrap, the real registered local
 * assets, the real SOURCE_ONLY producer, the real controlled adoption that commits
 * the rich source, the real working copy, the real approved 015 module and the real
 * reader.
 *
 * These are *component* cases: the review policy of the fixture is fixture data, the
 * session liveness callback is the fixture's own truth, and no human acceptance, no
 * native UI load and no NativeAck is claimed here.
 */
type Fixture=Awaited<ReturnType<typeof nativeSourceFixture>>;

/** The first read of one fixture. */
const readOnce=(f:Fixture):Promise<NativeExpectedSourceFacts>=>f.reader.readSource({binding:f.binding});

/** Attach the cleanup failures to a failure without hiding it. */
function withCleanup(error:unknown,failures:readonly unknown[]):never{
 if(failures.length>0&&typeof error==='object'&&error!==null){
  try{Object.defineProperty(error,'cleanup_errors',{value:failures,enumerable:false,configurable:true});}catch{/* the error carries what it can */}
 }
 throw error;
}

/**
 * Run a case over one fixture.
 *
 * Every owner is attempted even when an earlier one failed, and the primary failure
 * keeps its place with the cleanup failures attached to it.
 */
async function withFixture(body:(f:Fixture)=>Promise<void>):Promise<void>{
 await withFixtureOptions({},body);
}

/**
 * Run a case over one fixture of a given shape.
 *
 * The shape is the LAWFUL production input of the fixture: the plan without a
 * narration and the diagnostic video with a real audio track. Nothing of the
 * committed bytes is doctored - the pinned assembler and the created-source
 * publisher both support a null narration.
 */
async function withFixtureOptions(options:Parameters<typeof nativeSourceFixture>[0],body:(f:Fixture)=>Promise<void>):Promise<void>{
 const f=await nativeSourceFixture(options);
 let primary:unknown;
 try{await body(f);}
 catch(error){primary=error;}
 const failures:unknown[]=[];
 try{await f.close();}catch(error){failures.push(error);}
 if(primary!==undefined)withCleanup(primary,failures);
 if(failures.length>0){
  const failure=new Error('The case could not release everything');
  Object.defineProperty(failure,'cleanup_errors',{value:failures,enumerable:false,configurable:true});
  throw failure;
 }
}

/**
 * Run a case over one fixture and over the facts of one read.
 *
 * The delivered handles belong to the facts: they are released here, all of them, and
 * a failure of one is kept instead of being swallowed.
 */
async function withFacts(body:(facts:NativeExpectedSourceFacts,f:Fixture)=>Promise<void>):Promise<void>{
 await withFixture(async f=>{
  let facts:NativeExpectedSourceFacts|undefined;
  let primary:unknown;
  try{facts=await readOnce(f);await body(facts,f);}
  catch(error){primary=error;}
  const failures:unknown[]=[];
  if(facts)for(const handle of facts.retained){try{handle.close();}catch(error){failures.push(error);}}
  if(primary!==undefined)withCleanup(primary,failures);
  if(failures.length>0){
   const failure=new Error('The case could not release the delivered handles');
   Object.defineProperty(failure,'cleanup_errors',{value:failures,enumerable:false,configurable:true});
   throw failure;
  }
 });
}

/**
 * R02 — the closure of a proof does not survive a change of the working copy.
 *
 * The change is made after the read: the proof must not verify any more, and a
 * second read must refuse the copy that no longer is the committed source.
 */
it('R02 invalidates the old proof when the working copy changes under it',async()=>{
 await withFixture(async f=>{
  const facts=await readOnce(f);
  let handedOver=false;
  const expected=new OpenFilmNativeExpected({source:{readSource:async()=>{handedOver=true;return facts;}},
   loader:nativeExpectedLibraryLoader(),approved:nativeExpectedApprovalFromPin()});
  const failures:unknown[]=[];
  try{
   const proof=await expected.prove(f.binding);
   expect(expected.assertCurrent(f.binding,proof)).toBeDefined();
   // The same bytes, a different file: the content is equal and the identity is not.
   const film=join(f.workspaceRoot,'film.html');
   const bytes=readFileSync(film);
   const copy=join(f.workspaceRoot,'film-copy.html');
   writeFileSync(copy,bytes);
   renameSync(copy,film);
   // The old proof does not verify any more: the working copy is not the one it was
   // made against.
   expect(facts.verifyClosure()).toBe(false);
   // The expected service refuses the stale proof with its own reason. The old
   // interface states that reason in the MESSAGE of the refusal and not in a `code`
   // field, so the reason is asserted exactly - never as "some error".
   const stale=await Promise.resolve().then(()=>expected.assertCurrent(f.binding,proof)).then(()=>null,error=>error);
   expect(stale).toBeInstanceOf(Error);
   expect((stale as Error).message).toBe('NATIVE_EXPECTED_CLOSURE_MOVED');
   // A fresh round independently revalidates the committed bytes and may succeed.
   const next=await readOnce(f);
   expect(next.verifyClosure()).toBe(true);
   for(const handle of next.retained)handle.close();
   // Changed bytes are refused by the reader itself.
   writeFileSync(film,'not the committed movie');
   await expect(readOnce(f)).rejects.toMatchObject({code:'NATIVE_SOURCE_COPY_DRIFTED'});
  }finally{
   try{expected.close();}catch(error){failures.push(error);}
   if(!handedOver)for(const handle of facts.retained){try{handle.close();}catch(error){failures.push(error);}}
   if(failures.length>0)throw new Error('The case could not release everything: '+failures.map(String).join('; '));
  }
 });
},300000);

/**
 * R04 — the native project id, a forged probe cache and a file in flight.
 *
 * A *legitimate* cache is a different case: it must invalidate the proof that was
 * made before it appeared and be accepted by the next stable round.
 */

 await withFixture(async f=>{
  const media=readdirSync(join(f.workspaceRoot,'assets','video')).find(name=>name.endsWith('.mp4'));
  expect(media).toBeDefined();
  const mediaPath=`assets/video/${media!}`;
  const info=statSync(join(f.workspaceRoot,mediaPath));

  writeFileSync(join(f.workspaceRoot,'.film','id'),'someone-elses-project\n');

/**
 * R02 — the leases: parallel, repeated, foreign, copied and released.
 *
 * A lease is one acquisition. Releasing it releases that acquisition and nothing
 * else: the ref and every other lease of it stay usable, so a later stable round can
 * prove again on the same view.
 */
it('R02 keeps parallel leases independent and refuses foreign and copied tokens',async()=>{
 await withFixture(async f=>{
  const expectation={project_id:f.projectId,snapshot_id:f.binding.snapshot_id,
   source_digest:f.binding.source_digest,revision:f.binding.revision,native_project_id:f.binding.native_project_id};
  const first=f.workspace.acquireExpectedLease(f.workspaceRef,expectation);
  const second=f.workspace.acquireExpectedLease(f.workspaceRef,expectation);
  expect(second).not.toBe(first);
  expect(first.assertCurrent()).toBe(true);
  expect(second.assertCurrent()).toBe(true);
  // Releasing one lease does not revoke the ref or the other lease.
  first.release();
  expect(first.assertCurrent()).toBe(false);
  expect(second.assertCurrent()).toBe(true);
  // A repeated acquisition after a release still works: the view is unchanged.
  const third=f.workspace.acquireExpectedLease(f.workspaceRef,expectation);
  expect(third.assertCurrent()).toBe(true);
  third.release();
  // A copy of a lease is not a lease of this service.
  expect(f.workspace.isLease({...second})).toBe(false);
  expect(f.workspace.isLease(second)).toBe(true);
  // A lease of another service is not a lease of this one.
  const other=new CreatedNativeWorkspaceService(f.host.revisions,f.app.creation,f.app.committedSnapshots,
   {root:join(f.root,'other-workspaces'),protectedRoots:[join(f.root,'projects'),join(f.root,'snapshots'),vendorRoot]});
  try{
   const view=await other.prepare(f.projectId,f.owner);
   const ref='foreign-ws';
   const registration=other.registerWorkspaceRef(view,ref,{project_id:f.projectId,principal_id:'owner',
    transaction_id:f.transactionId,snapshot_id:view.snapshot_id,source_digest:view.source_digest,
    revision:view.revision,native_project_id:f.binding.native_project_id});
   expect(registration.generation).toBeGreaterThan(0);
   const foreign=other.acquireExpectedLease(ref,{project_id:f.projectId,snapshot_id:view.snapshot_id,
    source_digest:view.source_digest,revision:view.revision,native_project_id:f.binding.native_project_id});
   // The foreign lease really exists and really is current: the refusal below is
   // about who issued it, not about a lease that was never made.
   expect(foreign.assertCurrent()).toBe(true);
   expect(other.isLease(foreign)).toBe(true);
   expect(f.workspace.isLease(foreign)).toBe(false);
  }finally{
   await other.disposeAfterNative();
  }
 });
},300000);

/**
 * R02 — the whole closure travels with the handles and is released with the last one.
 */
it('R02 releases the whole closure only after the last retained leaf',async()=>{
 await withFacts(async facts=>{
  expect(facts.retained.length).toBeGreaterThan(1);
  expect(facts.verifyClosure()).toBe(true);
  const handles=[...facts.retained];
  // Every leaf but the last: the closure is still whole, so the proof still verifies.
  for(const handle of handles.slice(0,-1))handle.close();
  expect(facts.verifyClosure()).toBe(true);
  // The last leaf: the closure goes with it.
  handles[handles.length-1]!.close();
  expect(facts.verifyClosure()).toBe(false);
 });
},300000);

/**
 * R06 — the placement of every clip is the film's own, and the plan agrees with it.
 *
 * The film's own placement is read again, independently, from the committed bytes.
 */
it('R06 places every clip where the committed film places it',async()=>{
 await withFacts(async (facts,f)=>{
  const film=readFileSync(join(f.sourceRoot,'film.html'),'utf8');

/**
 * A stand-in for the media tool's child.
 *
 * It is an INJECTION: it never carries a tool answer, and it exists so that the
 * production runner takes its real failure branch. The runner, the parser and the
 * child bookkeeping are the production ones.
 */
function injectedToolChild(fault:'spawn'|'nonzero'|'malformed'|'output'|'hang'):unknown{
 const handlers:{close?:(code:number|null,signal:NodeJS.Signals|null)=>void}={};
 const later=(fn:()=>void)=>{setTimeout(fn,0);};
 const stdout={on(_event:'data',listener:(chunk:Buffer)=>void){
  if(fault==='malformed')later(()=>listener(Buffer.from('this is not json')));
  if(fault==='output')later(()=>listener(Buffer.alloc(5*1024*1024)));
 }};
 const stderr={on(_event:'data',listener:(chunk:Buffer)=>void){void listener;}};
 const child={
  pid:4242,exitCode:null,signalCode:null,stdout,stderr,
  on(event:string,listener:(...args:never[])=>void){
   if(event==='spawn')later(()=>(listener as ()=>void)());
   if(event==='error'&&fault==='spawn')later(()=>(listener as (error:unknown)=>void)(new Error('injected spawn failure')));
   return child;
  },
  once(event:string,listener:(...args:never[])=>void){
   if(event==='close'){
    handlers.close=listener as (code:number|null,signal:NodeJS.Signals|null)=>void;
    if(fault==='nonzero')later(()=>handlers.close?.(1,null));
    else if(fault==='spawn')later(()=>handlers.close?.(null,null));
    else if(fault==='malformed'||fault==='output')later(()=>handlers.close?.(0,null));
    // The hang ends only when the runner kills the child.
   }
   return child;
  },
  kill(){
   if(fault==='hang')later(()=>handlers.close?.(null,'SIGKILL'));
   return true;
  },
 };
 return child;
}

/** The canonical JSON the reader digests with: sorted keys, no whitespace. */
const canonicalJsonOf=(value:unknown):string=>{
 if(value===null||typeof value!=='object')return JSON.stringify(value)??'null';
 if(Array.isArray(value))return '['+value.map(canonicalJsonOf).join(',')+']';
 const record=value as Record<string,unknown>;
 return '{'+Object.keys(record).sort().map(key=>JSON.stringify(key)+':'+canonicalJsonOf(record[key])).join(',')+'}';
};

/**
 * The upstream's own normalisation, written here from the upstream's rules, so that the
 * expected probe evidence is derived independently of the reader's own function.
 */
const upstreamNormalized=(answer:unknown):Record<string,unknown>=>{
 const raw=answer as {format?:{duration?:unknown;format_name?:unknown};streams?:{codec_type?:string;width?:number;height?:number;avg_frame_rate?:string;codec_name?:string;pix_fmt?:string;side_data_list?:{rotation?:number}[];tags?:{rotate?:number}}[]};
 const streams=raw.streams??[];
 const video=streams.find(stream=>stream.codec_type==='video');
 const audio=streams.some(stream=>stream.codec_type==='audio');
 const duration=Number(raw.format?.duration);
 const kept=Number.isFinite(duration)&&duration>0&&!/image|png|mjpeg/.test(String(raw.format?.format_name??''))?duration:undefined;
 const [numerator,denominator]=String(video?.avg_frame_rate??'0/1').split('/').map(Number);
 const fps=numerator&&denominator?Math.round((numerator/denominator)*100)/100:undefined;
 const turn=Number(video?.side_data_list?.find(entry=>entry.rotation!=null)?.rotation??video?.tags?.rotate??0);
 const quarter=Math.abs(Math.round(turn/90))%2===1;
 const width=video?.width;
 const height=video?.height;
 return {
  ...(kept===undefined?{}:{duration:kept}),
  ...(video===undefined?{}:quarter?{width:height,height:width}:{width,height}),
  ...(fps===undefined?{}:{fps}),
  ...(video?.codec_name?{codec:String(video.codec_name),pixfmt:String(video.pix_fmt??'')}:{}),
  audio,
 };
};

/**
 * The SHA256 of a file, computed HERE and independently of the reader.
 *
 * It is a stream of 64 KiB, never the whole file in memory: the fact it is compared
 * with is a fact about the bytes on the disk, and the comparison must not depend on
 * the reader's own hashing.
 */
function streamingSha256(absolute:string):string{
 const hash=createHash('sha256');
 const descriptor=openSync(absolute,'r');
 try{
  const buffer=Buffer.allocUnsafe(64*1024);
  let position=0;
  for(;;){
   const read=readSync(descriptor,buffer,0,buffer.length,position);
   if(read===0)break;
   hash.update(buffer.subarray(0,read));
   position+=read;
  }
 }finally{closeSync(descriptor);}
 return hash.digest('hex');
}

/**
 * The cause chain of a failure, as far as it goes.
 *
 * A fresh reason keeps the original failure as its cause, so the chain is where the
 * original reason is found.
 */
function causeChainOf(error:unknown):unknown[]{
 const chain:unknown[]=[];
 let current:unknown=error;
 while(current!==undefined&&current!==null&&chain.length<8){
  chain.push(current);
  current=(current as {cause?:unknown}).cause;
 }
 return chain;
}

/**
 * P07 — the named primary and EVERY cleanup cause of a real failing unpack.
 *
 * The failure is the production one: the retained tree cannot be unpacked, and the
 * descriptors that were already taken are released while the failure is being
 * handled. The primary is the syscall failure that stopped the walk - it is named,
 * and it keeps its own place. Three INDEPENDENT releases fail as well - the walk's
 * own child descriptor on the nested unwind, the ancestry above the root and the
 * root itself. Every descriptor is closed for real before its failure is reported,
 * and every allocation is released exactly once.
 *
 * The child's own release really fails here: the case does not let it succeed in
 * order to avoid the very defect it is about.
 */
it('P07 keeps the named primary and every cleanup cause of the real unpack path',async()=>{
 const root=closureTree();
 const resolved=resolve(root);
 const primary=Object.assign(new Error('injected primary: the retained child directory could not be opened'),{code:'EIO'});
 const pathOfFd=new Map<number,string>();
 const opened:string[]=[];
 const closed:string[]=[];
 const cleanupFailures=new Map<string,Error>();
 let fired=0;
 let failure:unknown;
 try{
  await RetainedNativeClosure.open(root,{},{
   open:(path:string,flags:number)=>{
    // The named primary: the walk cannot open the directory below the media.
    if(path.endsWith(sep+'assets'+sep+'vo')){fired+=1;throw primary;}
    const descriptor=openSync(path,flags);
    pathOfFd.set(descriptor,path);
    opened.push(path);
    return descriptor;
   },
   close:(descriptor:number)=>{
    const path=pathOfFd.get(descriptor)??`#${descriptor}`;
    closed.push(path);
    // The descriptor is closed FIRST: the failure is reported afterwards, so no
    // descriptor number is left open and none is closed twice.
    closeSync(descriptor);
    // Three NAMED allocations fail their release: the walk's own child on the nested
    // unwind, the ancestry above the root and the root itself.
    if(path===resolved+sep+'assets'||path===dirname(resolved)||path===resolved){
     const cleanup=new Error(`injected release failure of ${path}`);
     cleanupFailures.set(path,cleanup);
     throw cleanup;
    }
   },
  });
 }catch(error){failure=error;}
 try{
  // The primary really fired: no case may pass on a failure that never happened.
  expect(fired).toBe(1);
  expect(failure).toBeDefined();
  const causes=(failure as {cleanup_errors?:unknown[]}).cleanup_errors;
  expect(Array.isArray(causes)).toBe(true);
  // The named primary keeps its place: it is the FIRST cause, it is the very object
  // that was injected, and the cleanup that followed it did not overwrite it.
  expect(causes![0]).toBe(primary);
  expect((primary as {code?:string}).code).toBe('EIO');
  // Each of the THREE NAMED releases really failed, and each failure is kept.
  expect(cleanupFailures.size).toBe(3);
  expect([...cleanupFailures.keys()].sort()).toEqual([resolved+sep+'assets',dirname(resolved),resolved].sort());
  for(const cleanup of cleanupFailures.values())expect(causes).toContain(cleanup);
  expect(causes!.length).toBe(cleanupFailures.size+1);
  // The refusal is the production one: the syscall error is kept as its cause and
  // the reason of the refusal is the production reason.
  expect((failure as {code?:string}).code).toBe('NATIVE_CLOSURE_UNSUPPORTED_TYPE');
  // Every allocation that was taken was released exactly once, and no descriptor
  // was released twice. The paths are the real ones, so a number that the operating
  // system hands out again for another allocation is not a retry.
  expect(opened.length).toBeGreaterThan(0);
  expect(closed.length).toBe(opened.length);
  expect(new Set(closed).size).toBe(closed.length);
  expect([...closed].sort()).toEqual([...opened].sort());
 }finally{
  rmSync(root,{recursive:true,force:true});
 }
 // The uncertainty is sticky: a release that failed latches the closure, and a
 // closure that cannot say what it holds can no longer prove anything.
 const second=closureTree();
 let closure:RetainedNativeClosure|undefined;
 const released:number[]=[];
 let armed=false;
 try{
  closure=await RetainedNativeClosure.open(second,{},{close:(descriptor:number)=>{
   closeSync(descriptor);
   released.push(descriptor);
   if(armed){armed=false;throw new Error('injected final release failure');}
  }});
  await closure.retainFile('film.html');
  // Only the final release is counted: the walk's own releases are done.
  released.length=0;
  armed=true;
  let releaseFailure:unknown;
  try{closure.close();}catch(error){releaseFailure=error;}
  expect(releaseFailure).toBeDefined();
  expect((releaseFailure as {code?:string}).code).toBe('NATIVE_CLOSURE_CLEANUP_FAILED');
  // Every outstanding allocation was attempted, each exactly once, even though the
  // first of them failed.
  expect(released.length).toBeGreaterThan(0);
  expect(new Set(released).size).toBe(released.length);
  // The failure is reported once and never twice.
  expect(closure.cleanupFailures.length).toBe(1);
  // A repeated release does not retry the allocation whose release failed.
  const attempted=released.length;
  expect(()=>closure!.close()).not.toThrow();
  expect(released.length).toBe(attempted);
  expect(closure.cleanupFailures.length).toBe(1);
 }finally{
  try{closure?.close();}catch{/* the closure reports its own failures */}
  rmSync(second,{recursive:true,force:true});
 }
},60000);
